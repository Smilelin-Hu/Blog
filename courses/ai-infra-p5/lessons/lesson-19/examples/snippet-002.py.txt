import unittest

from sft_compiler import SPECIAL, compile_messages, fixture, pack


class SFTCompilerTest(unittest.TestCase):
    def test_user_literal_does_not_become_control(self) -> None:
        compiled = compile_messages("x", fixture(), "all_assistant")
        user_positions = [i for i, role in enumerate(compiled.roles) if role == "user"]
        self.assertTrue(user_positions)
        self.assertTrue(all(compiled.input_ids[i] != SPECIAL["<AST>"] for i in user_positions))

    def test_tool_result_is_not_supervised(self) -> None:
        compiled = compile_messages("x", fixture(), "all_assistant")
        tool_positions = [i for i, role in enumerate(compiled.roles) if role == "tool"]
        self.assertTrue(all(compiled.labels[i] == -100 for i in tool_positions))

    def test_last_assistant_is_subset(self) -> None:
        all_turns = compile_messages("x", fixture(), "all_assistant")
        last = compile_messages("x", fixture(), "last_assistant")
        self.assertLess(last.supervised_tokens, all_turns.supervised_tokens)

    def test_pack_padding_is_ignored(self) -> None:
        compiled = compile_messages("x", fixture(), "all_assistant")
        packed = pack([compiled], len(compiled.input_ids) + 8)["packs"][0]
        self.assertEqual(packed["labels"][-8:], [-100] * 8)
        self.assertEqual(packed["segment_ids"][-8:], [-1] * 8)

    def test_pack_blocks_cross_sample_attention(self) -> None:
        first = compile_messages("a", fixture(), "last_assistant")
        second = compile_messages("b", fixture(), "last_assistant")
        packed = pack([first, second], len(first.input_ids) + len(second.input_ids))["packs"][0]
        split = len(first.input_ids)
        self.assertEqual(packed["attention_blocks"][split][split - 1], 0)
        self.assertEqual(packed["position_ids"][split], 0)


if __name__ == "__main__":
    unittest.main()

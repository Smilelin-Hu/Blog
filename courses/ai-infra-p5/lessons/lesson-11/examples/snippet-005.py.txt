from __future__ import annotations

import argparse
import json
import math
import statistics
import time

try:
    import torch
except Exception as exc:
    print(json.dumps({"status": "skipped", "reason": f"PyTorch import failed: {exc}"}))
    raise SystemExit(0)


def make_signatures(
    num_documents: int,
    signature_length: int,
    duplicate_fraction: float,
    mutation_fraction: float,
    seed: int,
) -> torch.Tensor:
    generator = torch.Generator(device="cpu").manual_seed(seed)
    signatures = torch.randint(
        low=0,
        high=2**31 - 1,
        size=(num_documents, signature_length),
        dtype=torch.int64,
        generator=generator,
    )
    duplicate_count = int(num_documents * duplicate_fraction)
    for index in range(1, duplicate_count):
        source = index % max(1, duplicate_count // 8)
        signatures[index].copy_(signatures[source])
        mutation_count = max(1, int(signature_length * mutation_fraction))
        positions = torch.randperm(signature_length, generator=generator)[:mutation_count]
        signatures[index, positions] = torch.randint(
            0,
            2**31 - 1,
            (mutation_count,),
            dtype=torch.int64,
            generator=generator,
        )
    return signatures


def make_pairs(
    num_documents: int,
    num_pairs: int,
    duplicate_documents: int,
    seed: int,
) -> torch.Tensor:
    generator = torch.Generator(device="cpu").manual_seed(seed)
    near_count = num_pairs // 2
    left_near = torch.randint(0, max(1, duplicate_documents // 8), (near_count,), generator=generator)
    offsets = torch.randint(1, 8, (near_count,), generator=generator)
    right_near = (left_near + offsets) % max(1, duplicate_documents)
    random_count = num_pairs - near_count
    left_random = torch.randint(0, num_documents, (random_count,), generator=generator)
    right_random = torch.randint(0, num_documents, (random_count,), generator=generator)
    return torch.stack((
        torch.cat((left_near, left_random)),
        torch.cat((right_near, right_random)),
    ), dim=1)


def compare_batches(
    signatures: torch.Tensor,
    pairs: torch.Tensor,
    device: torch.device,
    batch_pairs: int,
    threshold: float,
) -> tuple[torch.Tensor, float, float]:
    signatures_device = signatures.to(device, non_blocking=True)
    torch.cuda.synchronize(device)
    torch.cuda.reset_peak_memory_stats(device)
    outputs = []
    started = time.perf_counter()
    for start in range(0, pairs.shape[0], batch_pairs):
        pair_batch = pairs[start:start + batch_pairs].to(device, non_blocking=True)
        left = signatures_device.index_select(0, pair_batch[:, 0])
        right = signatures_device.index_select(0, pair_batch[:, 1])
        estimates = left.eq(right).float().mean(dim=1)
        outputs.append(torch.nonzero(estimates >= threshold, as_tuple=False).squeeze(1).cpu() + start)
    torch.cuda.synchronize(device)
    wall = time.perf_counter() - started
    peak = torch.cuda.max_memory_allocated(device) / 1024**2
    return torch.cat(outputs) if outputs else torch.empty(0, dtype=torch.int64), wall, peak


def cpu_reference(
    signatures: torch.Tensor,
    pairs: torch.Tensor,
    threshold: float,
) -> torch.Tensor:
    outputs = []
    for index, (left_id, right_id) in enumerate(pairs.tolist()):
        estimate = signatures[left_id].eq(signatures[right_id]).float().mean().item()
        if estimate >= threshold:
            outputs.append(index)
    return torch.tensor(outputs, dtype=torch.int64)


def percentile(values: list[float], fraction: float) -> float:
    ordered = sorted(values)
    position = min(len(ordered) - 1, math.ceil(len(ordered) * fraction) - 1)
    return ordered[position]


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--documents", type=int, default=200_000)
    parser.add_argument("--pairs", type=int, default=2_000_000)
    parser.add_argument("--signature-length", type=int, default=128)
    parser.add_argument("--batch-pairs", type=int, default=65_536)
    parser.add_argument("--duplicate-fraction", type=float, default=0.10)
    parser.add_argument("--mutation-fraction", type=float, default=0.10)
    parser.add_argument("--threshold", type=float, default=0.80)
    parser.add_argument("--repeat", type=int, default=5)
    args = parser.parse_args()

    if not torch.cuda.is_available():
        print(json.dumps({"status": "skipped", "reason": "CUDA GPU is not available"}))
        return

    device = torch.device("cuda:0")
    signatures = make_signatures(
        args.documents,
        args.signature_length,
        args.duplicate_fraction,
        args.mutation_fraction,
        seed=20260821,
    )
    duplicate_documents = int(args.documents * args.duplicate_fraction)
    pairs = make_pairs(args.documents, args.pairs, duplicate_documents, seed=17)

    reference_size = min(10_000, args.pairs)
    reference = cpu_reference(signatures, pairs[:reference_size], args.threshold)
    observed, _, _ = compare_batches(
        signatures,
        pairs[:reference_size],
        device,
        args.batch_pairs,
        args.threshold,
    )
    if not torch.equal(reference, observed):
        raise AssertionError("CUDA result differs from CPU reference")

    compare_batches(signatures, pairs[:min(args.pairs, 100_000)], device, args.batch_pairs, args.threshold)
    durations = []
    peaks = []
    selected = 0
    for _ in range(args.repeat):
        result, wall, peak = compare_batches(
            signatures, pairs, device, args.batch_pairs, args.threshold
        )
        durations.append(wall)
        peaks.append(peak)
        selected = result.numel()

    median = statistics.median(durations)
    report = {
        "status": "succeeded",
        "device": torch.cuda.get_device_name(0),
        "torch": torch.__version__,
        "cuda_runtime": torch.version.cuda,
        "configuration": vars(args),
        "correctness_pairs": reference_size,
        "selected_pairs": selected,
        "selected_fraction": selected / max(args.pairs, 1),
        "wall_seconds": durations,
        "median_wall_seconds": median,
        "p95_wall_seconds": percentile(durations, 0.95),
        "median_pairs_per_second": args.pairs / median,
        "peak_memory_mib": max(peaks),
    }
    print(json.dumps(report, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()

from __future__ import annotations

import argparse
import hashlib
import json
import os
import struct
import time
from pathlib import Path
from typing import Any

import torch
import torch.distributed as dist
from torch import nn


MAGIC = b"TKSHARD1"
HEADER = struct.Struct("<8sIIQ")
TOKEN = struct.Struct("<I")


def file_sha(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        while block := handle.read(1024 * 1024):
            digest.update(block)
    return digest.hexdigest()


def load_manifest(root: Path) -> dict[str, Any]:
    return json.loads((root / "manifest.json").read_text(encoding="utf-8"))


def load_entries(root: Path, manifest: dict[str, Any]) -> list[dict[str, Any]]:
    records = []
    for shard_number, shard in enumerate(manifest["shards"]):
        bin_path = root / shard["bin_file"]
        index_path = root / shard["index_file"]
        if file_sha(bin_path) != shard["bin_sha256"]:
            raise ValueError("bin digest mismatch")
        if file_sha(index_path) != shard["index_sha256"]:
            raise ValueError("index digest mismatch")
        index = json.loads(index_path.read_text(encoding="utf-8"))
        if index["parent_bin_sha256"] != shard["bin_sha256"]:
            raise ValueError("parent digest mismatch")
        with bin_path.open("rb") as handle:
            magic, version, dtype_code, total_tokens = HEADER.unpack(handle.read(HEADER.size))
        if magic != MAGIC or version != 1 or dtype_code != 1:
            raise ValueError("unsupported header")
        for value in index["entries"]:
            if value["token_offset"] + value["token_length"] > total_tokens:
                raise ValueError("index out of bounds")
            records.append({"shard_number": shard_number, **value})
    records.sort(
        key=lambda value: hashlib.sha256(
            f"2026:{value['sample_id']}".encode("utf-8")
        ).digest()
    )
    return records


def read_tokens(root: Path, manifest: dict[str, Any], record: dict[str, Any]) -> list[int]:
    shard = manifest["shards"][record["shard_number"]]
    byte_offset = HEADER.size + int(record["token_offset"]) * TOKEN.size
    byte_length = int(record["token_length"]) * TOKEN.size
    with (root / shard["bin_file"]).open("rb") as handle:
        handle.seek(byte_offset)
        payload = handle.read(byte_length)
    if len(payload) != byte_length:
        raise ValueError("short read")
    return [value[0] for value in struct.iter_unpack("<I", payload)]


def collate(sequences: list[list[int]], sequence_length: int, vocab_size: int) -> torch.Tensor:
    rows = []
    for sequence in sequences:
        values = [value % vocab_size for value in sequence[:sequence_length]]
        values.extend([0] * (sequence_length - len(values)))
        rows.append(values)
    return torch.tensor(rows, dtype=torch.long)


class TinyLM(nn.Module):
    def __init__(self, vocab_size: int, hidden: int) -> None:
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, hidden)
        self.projection = nn.Linear(hidden, vocab_size, bias=False)

    def forward(self, tokens: torch.Tensor) -> torch.Tensor:
        return self.projection(self.embedding(tokens))


def run(args: argparse.Namespace) -> None:
    rank = int(os.environ.get("RANK", "0"))
    local_rank = int(os.environ.get("LOCAL_RANK", "0"))
    world_size = int(os.environ.get("WORLD_SIZE", "1"))
    if not torch.cuda.is_available() or torch.cuda.device_count() < 1 or world_size < 1:
        if rank == 0:
            print(
                json.dumps(
                    {
                        "status": "skipped",
                        "reason": "requires torchrun with at least one visible CUDA device",
                        "cuda_devices": torch.cuda.device_count(),
                        "world_size": world_size,
                    },
                    ensure_ascii=False,
                )
            )
        return
    root = Path(args.dataset)
    manifest = load_manifest(root)
    manifest_digest = manifest["dataset_digest"]
    dist.init_process_group("nccl")
    torch.cuda.set_device(local_rank)
    device = torch.device("cuda", local_rank)
    gathered = [None for _ in range(world_size)]
    dist.all_gather_object(gathered, manifest_digest)
    if len(set(gathered)) != 1:
        raise RuntimeError("manifest differs across ranks")
    records = load_entries(root, manifest)
    global_batch = world_size * args.batch_size
    usable = len(records) - len(records) % global_batch
    deferred = len(records) - usable
    rank_records = records[:usable][rank::world_size]
    torch.manual_seed(args.seed)
    model = TinyLM(args.vocab_size, args.hidden).to(device)
    model = nn.parallel.DistributedDataParallel(model, device_ids=[local_rank])
    optimizer = torch.optim.AdamW(model.parameters(), lr=args.lr)
    criterion = nn.CrossEntropyLoss(ignore_index=0)
    torch.cuda.reset_peak_memory_stats(device)
    io_seconds = 0.0
    compute_start = time.perf_counter()
    local_ids = []
    local_tokens = 0
    steps = 0
    for offset in range(0, len(rank_records), args.batch_size):
        batch_records = rank_records[offset : offset + args.batch_size]
        io_start = time.perf_counter()
        sequences = [read_tokens(root, manifest, record) for record in batch_records]
        tokens = collate(sequences, args.sequence_length, args.vocab_size)
        io_seconds += time.perf_counter() - io_start
        tokens = tokens.to(device, non_blocking=True)
        logits = model(tokens[:, :-1])
        target = tokens[:, 1:]
        loss = criterion(logits.reshape(-1, args.vocab_size), target.reshape(-1))
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        optimizer.step()
        local_ids.extend(record["sample_id"] for record in batch_records)
        local_tokens += sum(min(len(sequence), args.sequence_length) for sequence in sequences)
        steps += 1
    torch.cuda.synchronize(device)
    elapsed = time.perf_counter() - compute_start
    metric = torch.tensor(
        [len(local_ids), local_tokens, steps, int(io_seconds * 1e9)],
        dtype=torch.long,
        device=device,
    )
    dist.all_reduce(metric, op=dist.ReduceOp.SUM)
    max_elapsed = torch.tensor([elapsed], dtype=torch.float64, device=device)
    max_memory = torch.tensor(
        [torch.cuda.max_memory_allocated(device)], dtype=torch.long, device=device
    )
    dist.all_reduce(max_elapsed, op=dist.ReduceOp.MAX)
    dist.all_reduce(max_memory, op=dist.ReduceOp.MAX)
    all_ids = [None for _ in range(world_size)] if rank == 0 else None
    dist.gather_object(local_ids, all_ids, dst=0)
    if rank == 0:
        flattened = [sample_id for values in (all_ids or []) for sample_id in (values or [])]
        expected = {record["sample_id"] for record in records[:usable]}
        coverage_ok = len(flattened) == len(set(flattened)) == len(expected) and set(flattened) == expected
        report = {
            "status": "passed" if coverage_ok else "failed",
            "manifest_digest": manifest_digest,
            "world_size": world_size,
            "usable_samples": usable,
            "deferred_samples": deferred,
            "consumed_samples": metric[0].item(),
            "consumed_tokens": metric[1].item(),
            "summed_rank_steps": metric[2].item(),
            "summed_io_seconds": metric[3].item() / 1e9,
            "max_elapsed_seconds": max_elapsed.item(),
            "tokens_per_second": metric[1].item() / max(max_elapsed.item(), 1e-9),
            "max_memory_bytes": max_memory.item(),
            "coverage_ok": coverage_ok,
        }
        print(json.dumps(report, ensure_ascii=False, sort_keys=True, indent=2))
        if not coverage_ok:
            raise SystemExit(1)
    dist.destroy_process_group()


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--dataset", required=True)
    parser.add_argument("--batch-size", type=int, default=2)
    parser.add_argument("--sequence-length", type=int, default=64)
    parser.add_argument("--vocab-size", type=int, default=4096)
    parser.add_argument("--hidden", type=int, default=128)
    parser.add_argument("--lr", type=float, default=1e-3)
    parser.add_argument("--seed", type=int, default=2026)
    run(parser.parse_args())


if __name__ == "__main__":
    main()

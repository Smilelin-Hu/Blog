from __future__ import annotations

import hashlib
import json
import math
import random


NAMES = ["web", "books", "code", "math"]


def normalize(values: list[float]) -> list[float]:
    total = sum(values)
    if total <= 0:
        raise ValueError("mixture has no mass")
    return [value / total for value in values]


def doremi_simulation(seed: int = 7) -> dict[str, object]:
    rng = random.Random(seed)
    weights = [0.25] * 4
    reference = [2.20, 2.00, 2.45, 2.70]
    learnability = [0.9, 0.7, 1.25, 1.5]
    cumulative = [0.0] * 4
    trajectory = []
    for step in range(1, 301):
        excess = []
        for i in range(4):
            progress = learnability[i] * math.sqrt(step * max(weights[i], 1e-4))
            proxy_loss = reference[i] + 0.85 - 0.055 * progress
            noise = rng.gauss(0.0, 0.025)
            excess.append(max(-0.4, proxy_loss - reference[i] + noise))
        eta = 0.22
        logits = [math.log(max(w, 1e-12)) + eta * r for w, r in zip(weights, excess)]
        pivot = max(logits)
        updated = normalize([math.exp(value - pivot) for value in logits])
        epsilon = 0.05
        weights = [(1 - epsilon) * w + epsilon / 4 for w in updated]
        for i, weight in enumerate(weights):
            cumulative[i] += weight
        if step % 50 == 0:
            trajectory.append({
                "step": step,
                "weights": [round(v, 5) for v in weights],
                "excess": [round(v, 5) for v in excess],
            })
    average = normalize([value / 300 for value in cumulative])
    return {"average": average, "trajectory": trajectory}


def sample_simplex(rng: random.Random, concentration: float) -> list[float]:
    raw = [rng.gammavariate(concentration, 1.0) for _ in NAMES]
    return normalize(raw)


def hidden_score(p: list[float]) -> float:
    # 越小越好；包含线性、二次和领域交互。
    target = [0.38, 0.17, 0.27, 0.18]
    quadratic = sum((a - b) ** 2 for a, b in zip(p, target))
    interaction = -0.22 * p[2] * p[3] + 0.16 * p[0] * p[1]
    return 1.7 + 2.4 * quadratic + interaction


def features(p: list[float]) -> list[float]:
    result = [1.0] + p[:-1]
    result += [value * value for value in p[:-1]]
    for i in range(len(p) - 1):
        for j in range(i + 1, len(p) - 1):
            result.append(p[i] * p[j])
    return result


def solve_linear(matrix: list[list[float]], vector: list[float]) -> list[float]:
    n = len(vector)
    augmented = [row[:] + [vector[i]] for i, row in enumerate(matrix)]
    for col in range(n):
        pivot = max(range(col, n), key=lambda r: abs(augmented[r][col]))
        augmented[col], augmented[pivot] = augmented[pivot], augmented[col]
        scale = augmented[col][col]
        if abs(scale) < 1e-12:
            raise ValueError("singular regression system")
        augmented[col] = [value / scale for value in augmented[col]]
        for row in range(n):
            if row == col:
                continue
            factor = augmented[row][col]
            augmented[row] = [
                a - factor * b for a, b in zip(augmented[row], augmented[col])
            ]
    return [augmented[i][-1] for i in range(n)]


def ridge_fit(xs: list[list[float]], ys: list[float], penalty: float) -> list[float]:
    width = len(xs[0])
    gram = [[0.0] * width for _ in range(width)]
    rhs = [0.0] * width
    for x, y in zip(xs, ys):
        for i in range(width):
            rhs[i] += x[i] * y
            for j in range(width):
                gram[i][j] += x[i] * x[j]
    for i in range(1, width):
        gram[i][i] += penalty
    return solve_linear(gram, rhs)


def predict(beta: list[float], p: list[float]) -> float:
    return sum(a * b for a, b in zip(beta, features(p)))


def spearman(actual: list[float], predicted: list[float]) -> float:
    def ranks(values: list[float]) -> list[int]:
        order = sorted(range(len(values)), key=lambda i: values[i])
        result = [0] * len(values)
        for rank, index in enumerate(order):
            result[index] = rank
        return result
    a, b = ranks(actual), ranks(predicted)
    n = len(a)
    distance = sum((x - y) ** 2 for x, y in zip(a, b))
    return 1.0 - 6.0 * distance / (n * (n * n - 1))


def regmix_simulation(seed: int = 11) -> dict[str, object]:
    rng = random.Random(seed)
    design = []
    for concentration in [0.25, 0.6, 1.5, 4.0]:
        design.extend(sample_simplex(rng, concentration) for _ in range(40))
    rng.shuffle(design)
    train, test = design[:128], design[128:]
    train_y = [hidden_score(p) + rng.gauss(0, 0.002) for p in train]
    beta = ridge_fit([features(p) for p in train], train_y, penalty=1e-4)
    test_actual = [hidden_score(p) for p in test]
    test_predicted = [predict(beta, p) for p in test]
    candidates = [sample_simplex(rng, 0.9) for _ in range(20_000)]
    feasible = [p for p in candidates if min(p) >= 0.03 and max(p) <= 0.60]
    ranked = sorted(feasible, key=lambda p: predict(beta, p))[:20]
    averaged = normalize([sum(p[i] for p in ranked) for i in range(4)])
    best_observed = min(hidden_score(p) for p in test)
    return {
        "test_spearman": round(spearman(test_actual, test_predicted), 5),
        "selected": [round(v, 5) for v in averaged],
        "selected_actual_score": round(hidden_score(averaged), 6),
        "best_test_score": round(best_observed, 6),
        "candidate_count": len(feasible),
    }


def digest(value: object) -> str:
    payload = json.dumps(value, sort_keys=True, separators=(",", ":")).encode()
    return hashlib.sha256(payload).hexdigest()


def main() -> None:
    result = {"doremi": doremi_simulation(), "regmix": regmix_simulation()}
    second = {"doremi": doremi_simulation(), "regmix": regmix_simulation()}
    assert digest(result) == digest(second)
    assert result["regmix"]["test_spearman"] > 0.8
    print(json.dumps({"status": "passed", **result}, indent=2))


if __name__ == "__main__":
    main()

from __future__ import annotations

import argparse
import hashlib
import json
import os
import random
import time
from pathlib import Path
from typing import Dict, List, Tuple


def emit(value: Dict[str, object]) -> None:
    print(json.dumps(value, ensure_ascii=False), flush=True)


try:
    import torch
    import torch.distributed as dist
    import torch.nn as nn
    from torch.nn.parallel import DistributedDataParallel as DDP
except Exception as exc:
    emit({"status": "skipped", "reason": "pytorch_import_failed", "detail": str(exc)})
    raise SystemExit(0)


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


class TinyLM(nn.Module):
    def __init__(self, vocab_size: int, sequence_length: int) -> None:
        super().__init__()
        hidden = 192
        self.token = nn.Embedding(vocab_size, hidden)
        self.position = nn.Embedding(sequence_length, hidden)
        layer = nn.TransformerEncoderLayer(
            hidden, 6, hidden * 4, dropout=0.0, batch_first=True, norm_first=True
        )
        self.layers = nn.TransformerEncoder(layer, 4)
        self.norm = nn.LayerNorm(hidden)
        self.head = nn.Linear(hidden, vocab_size, bias=False)
        self.head.weight = self.token.weight

    def forward(self, ids: torch.Tensor) -> torch.Tensor:
        length = ids.shape[1]
        positions = torch.arange(length, device=ids.device).unsqueeze(0)
        hidden = self.token(ids) + self.position(positions)
        mask = torch.triu(
            torch.full((length, length), float("-inf"), device=ids.device),
            diagonal=1,
        )
        return self.head(self.norm(self.layers(hidden, mask=mask, is_causal=True)))


def prepare(root: Path, seed: int, sequence_length: int, vocab_size: int) -> None:
    root.mkdir(parents=True, exist_ok=True)
    sources = [
        {
            "source_id": "source-tech-docs",
            "agreement_version": 1,
            "status": "active",
            "allowed_actions": ["train", "evaluate"],
            "allowed_purposes": ["internal_product"],
            "allowed_regions": ["us", "jp"],
            "allowed_distributions": ["internal_only"],
        },
        {
            "source_id": "source-synthetic-anchor",
            "agreement_version": 3,
            "status": "active",
            "allowed_actions": ["train"],
            "allowed_purposes": ["internal_product"],
            "allowed_regions": ["us"],
            "allowed_distributions": ["internal_only"],
        },
        {
            "source_id": "source-community-eval",
            "agreement_version": 1,
            "status": "active",
            "allowed_actions": ["evaluate"],
            "allowed_purposes": ["internal_research"],
            "allowed_regions": ["us"],
            "allowed_distributions": ["internal_only"],
        },
    ]
    objects = []
    for index, source in enumerate(sources):
        generator = torch.Generator().manual_seed(seed + index)
        starts = torch.randint(4, vocab_size, (2048, 1), generator=generator)
        steps = torch.randint(1, 17, (2048, 1), generator=generator)
        positions = torch.arange(sequence_length + 1).unsqueeze(0)
        tokens = 4 + ((starts - 4 + positions * steps) % (vocab_size - 4))
        path = root / f"source-{index:02d}.pt"
        torch.save(tokens.to(torch.int32), path)
        objects.append({
            **source,
            "path": path.name,
            "sha256": sha256_file(path),
            "rows": len(tokens),
            "weight": 0.7 if index == 0 else 0.3,
        })
    raw = {"schema": "source-policy-mixture-v1", "objects": objects}
    raw["registry_snapshot_id"] = "registry:" + hashlib.sha256(
        json.dumps(raw, sort_keys=True).encode("utf-8")
    ).hexdigest()
    (root / "registry.json").write_text(
        json.dumps(raw, ensure_ascii=False, indent=2) + "\n", encoding="utf-8"
    )


def decide(item: Dict[str, object], request: Dict[str, str]) -> Tuple[str, List[str]]:
    reasons = []
    if item["status"] != "active":
        reasons.append("source_not_active")
    if request["action"] not in item["allowed_actions"]:
        reasons.append("action_not_allowed")
    if request["purpose"] not in item["allowed_purposes"]:
        reasons.append("purpose_not_allowed")
    if request["region"] not in item["allowed_regions"]:
        reasons.append("region_not_allowed")
    if request["distribution"] not in item["allowed_distributions"]:
        reasons.append("distribution_not_allowed")
    return ("deny" if reasons else "allow"), reasons


def compile_mixture(root: Path, region: str) -> Dict[str, object]:
    registry = json.loads((root / "registry.json").read_text(encoding="utf-8"))
    request = {
        "action": "train",
        "purpose": "internal_product",
        "region": region,
        "distribution": "internal_only",
    }
    allowed = []
    decisions = []
    for item in registry["objects"]:
        result, reasons = decide(item, request)
        decisions.append({
            "source_id": item["source_id"],
            "agreement_version": item["agreement_version"],
            "result": result,
            "reasons": reasons,
        })
        if result == "allow":
            path = root / item["path"]
            if sha256_file(path) != item["sha256"]:
                raise RuntimeError(f"hash mismatch: {item['path']}")
            allowed.append({
                "source_id": item["source_id"],
                "path": item["path"],
                "sha256": item["sha256"],
                "weight": item["weight"],
            })
    if not allowed:
        raise PermissionError("no source is allowed for requested training context")
    total = sum(float(item["weight"]) for item in allowed)
    for item in allowed:
        item["weight"] = float(item["weight"]) / total
    body = {
        "request": request,
        "registry_snapshot_id": registry["registry_snapshot_id"],
        "decisions": decisions,
        "allowed_objects": allowed,
    }
    body["mixture_commit_id"] = "mixture:" + hashlib.sha256(
        json.dumps(body, sort_keys=True).encode("utf-8")
    ).hexdigest()
    (root / "mixture.json").write_text(
        json.dumps(body, ensure_ascii=False, indent=2) + "\n", encoding="utf-8"
    )
    return body


def setup() -> Tuple[int, int, int]:
    if not torch.cuda.is_available():
        emit({"status": "skipped", "reason": "cuda_unavailable"})
        raise SystemExit(0)
    if torch.cuda.device_count() < 1:
        emit({"status": "skipped", "reason": "requires_two_cuda_gpus", "detected": torch.cuda.device_count()})
        raise SystemExit(0)
    if any(key not in os.environ for key in ("RANK", "WORLD_SIZE", "LOCAL_RANK")):
        emit({"status": "skipped", "reason": "launch_with_torchrun"})
        raise SystemExit(0)
    rank = int(os.environ["RANK"])
    world = int(os.environ["WORLD_SIZE"])
    local = int(os.environ["LOCAL_RANK"])
    if world < 1:
        if rank == 0:
            emit({"status": "skipped", "reason": "world_size_must_be_positive", "world_size": world})
        raise SystemExit(0)
    torch.cuda.set_device(local)
    dist.init_process_group("nccl")
    return rank, world, local


def train(args: argparse.Namespace, mixture: Dict[str, object]) -> None:
    rank, world, local = setup()
    device = torch.device("cuda", local)
    torch.manual_seed(args.seed)
    datasets = []
    weights = []
    for item in mixture["allowed_objects"]:
        datasets.append(torch.load(args.root / item["path"], weights_only=True).long())
        weights.append(float(item["weight"]))
    model = DDP(TinyLM(args.vocab_size, args.sequence_length).to(device), device_ids=[local])
    optimizer = torch.optim.AdamW(model.parameters(), lr=3e-4)
    loss_fn = nn.CrossEntropyLoss()
    rng = random.Random(args.seed + rank)
    recent: List[float] = []
    source_counts = [0 for _ in datasets]
    started = time.perf_counter()

    for _ in range(args.steps):
        source_index = rng.choices(range(len(datasets)), weights=weights, k=1)[0]
        data = datasets[source_index]
        indices = [rng.randrange(len(data)) for _ in range(args.batch_size)]
        batch = data[indices].to(device, non_blocking=True)
        optimizer.zero_grad(set_to_none=True)
        logits = model(batch[:, :-1])
        labels = batch[:, 1:]
        loss = loss_fn(logits.reshape(-1, args.vocab_size), labels.reshape(-1))
        loss.backward()
        optimizer.step()
        recent.append(loss.item())
        recent = recent[-20:]
        source_counts[source_index] += labels.numel()

    counts = torch.tensor(source_counts, device=device, dtype=torch.long)
    dist.all_reduce(counts, op=dist.ReduceOp.SUM)
    torch.cuda.synchronize(device)
    if rank == 0:
        checkpoint = {
            "model": model.module.state_dict(),
            "evidence": {
                "mixture_commit_id": mixture["mixture_commit_id"],
                "registry_snapshot_id": mixture["registry_snapshot_id"],
                "request": mixture["request"],
                "decisions": mixture["decisions"],
                "allowed_objects": mixture["allowed_objects"],
                "source_token_counts": counts.cpu().tolist(),
                "world_size": world,
            },
        }
        torch.save(checkpoint, args.checkpoint)
        emit({
            "status": "completed",
            "mixture_commit_id": mixture["mixture_commit_id"],
            "final_train_loss": sum(recent) / len(recent),
            "source_token_counts": counts.cpu().tolist(),
            "checkpoint": str(args.checkpoint),
            "gpu_0": torch.cuda.get_device_name(0),
            "gpu_1": torch.cuda.get_device_name(1),
            "elapsed_seconds": time.perf_counter() - started,
            "proof_limit": "real DDP policy-gated input with fictional sources",
        })
    dist.destroy_process_group()


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser()
    parser.add_argument("--root", type=Path, default=Path("source_policy_data"))
    parser.add_argument("--prepare", action="store_true")
    parser.add_argument("--region", default="us")
    parser.add_argument("--steps", type=int, default=120)
    parser.add_argument("--batch-size", type=int, default=16)
    parser.add_argument("--sequence-length", type=int, default=128)
    parser.add_argument("--vocab-size", type=int, default=512)
    parser.add_argument("--seed", type=int, default=20260821)
    parser.add_argument("--checkpoint", type=Path, default=Path("source_policy_checkpoint.pt"))
    return parser.parse_args()


def main() -> None:
    args = parse_args()
    rank = int(os.environ.get("RANK", "0"))
    if args.prepare and rank == 0:
        prepare(args.root, args.seed, args.sequence_length, args.vocab_size)
    if "WORLD_SIZE" in os.environ:
        while not (args.root / "registry.json").exists():
            time.sleep(0.05)
    try:
        mixture = compile_mixture(args.root, args.region)
    except Exception as exc:
        if rank == 0:
            emit({"status": "failed", "reason": "policy_or_manifest_compilation_failed", "detail": str(exc)})
        raise SystemExit(2)
    train(args, mixture)


if __name__ == "__main__":
    main()

from __future__ import annotations

import argparse
import json
from dataclasses import dataclass


@dataclass
class State:
    trainer_version: int = 0
    rollout_version: int = 0
    queue: float = 0.0
    generated: float = 0.0
    accepted: float = 0.0
    dropped: float = 0.0
    sync_seconds: float = 0.0


def simulate(
    duration: int,
    update_every: int,
    sync_every: int,
    weight_gb: float,
    bandwidth_gbps: float,
    workers: int,
    tokens_per_worker_second: float,
    tokens_per_rollout: float,
    max_staleness: int,
) -> dict[str, float]:
    state = State()
    syncing_until = -1
    next_sync = sync_every
    for second in range(duration):
        if second > 0 and second % update_every == 0:
            state.trainer_version += 1
        if second >= next_sync and syncing_until < second:
            transfer = max(1, int(weight_gb * 8 / max(0.01, bandwidth_gbps)))
            syncing_until = second + transfer
            state.sync_seconds += transfer
            next_sync = second + sync_every
        if syncing_until == second:
            state.rollout_version = state.trainer_version
        if second < syncing_until:
            continue
        new_rollouts = workers * tokens_per_worker_second / tokens_per_rollout
        state.generated += new_rollouts
        staleness = state.trainer_version - state.rollout_version
        if staleness <= max_staleness:
            state.accepted += new_rollouts
        else:
            state.dropped += new_rollouts
    return {
        "trainer_version": state.trainer_version,
        "rollout_version": state.rollout_version,
        "generated": round(state.generated, 2),
        "accepted": round(state.accepted, 2),
        "dropped": round(state.dropped, 2),
        "accepted_per_second": round(state.accepted / duration, 4),
        "sync_time_fraction": round(state.sync_seconds / duration, 4),
    }


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--duration", type=int, default=3600)
    parser.add_argument("--weight-gb", type=float, default=14.0)
    parser.add_argument("--bandwidth-gbps", type=float, default=25.0)
    args = parser.parse_args()
    rows = []
    for sync_every in [30, 60, 120, 300, 600]:
        rows.append({
            "sync_every": sync_every,
            **simulate(
                duration=args.duration,
                update_every=60,
                sync_every=sync_every,
                weight_gb=args.weight_gb,
                bandwidth_gbps=args.bandwidth_gbps,
                workers=4,
                tokens_per_worker_second=600.0,
                tokens_per_rollout=1200.0,
                max_staleness=2,
            ),
        })
    print(json.dumps({"config": vars(args), "results": rows}, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
